% kacho-minimal.tex -- smallest working example of the kacho engine (M1 demo layout).
% Build: make            (or: TEXINPUTS=$PWD/tex//: LUAINPUTS=$PWD/tex//: lualatex -output-directory=build examples/kacho-minimal.tex)
% It exercises two-sided mirroring (odd/even pages), a per-page override,
% \kachoOff/\kachoOn and a partial last page.
\documentclass[11pt,twoside]{article}
\usepackage[a5paper,inner=15mm,outer=38mm,top=22mm,bottom=26mm]{geometry}
\usepackage{lipsum}
\usepackage[seed=7, palette=sumi, layout=demo, ground=washi]{kacho}
\title{kacho: minimal example}
\author{procedural kach\=o-e margins for Lua\LaTeX}
\date{}
\begin{document}
\maketitle
\section*{A branch and a wash}
\lipsum[1-3]

\kachoThisPage{seed=11, palette=hanga}% different composition and palette on the page containing this marker
\lipsum[4-6]
\clearpage
\kachoOff
\section*{An undecorated page}
This page is intentionally plain (\verb|\kachoOff|).
\clearpage
\kachoOn
\section*{Decoration resumes}
\lipsum[7-8]
\end{document}
