% kacho-animated.tex: optional 24-frame procedural growth animation (M5, Feature 31).
%
% Build commands:
%   make animated         -> build/kacho-animated.pdf (interactive PDF animation, poster=last)
%   make view-animated    -> open in viewer (or render PNG preview if headless)
%   make animated-frames  -> build/kacho-animated-frames.pdf + build/png/kacho-animated-frame-*.png
%
% Debug / frame-inspection switches:
%   \def\kachoAnimFrames{1} (or \def\framespages{1} or KACHO_ANIM_FRAMES=1)
%     emits one frame per page with the same 220x280 bp clipping box as animateinline.
\documentclass{article}
\usepackage[paperwidth=270bp,paperheight=345bp,top=16bp,bottom=16bp,left=25bp,right=25bp]{geometry}
\usepackage{pgfcore}
\usepackage{animate}
\pagestyle{empty}
\setlength{\parindent}{0pt}

\directlua{
  anim = require("kacho-anim")
  anim.plan(7, 220, 280, {
    box_x0 = 25,
    box_y0 = 28,
    textblock = { 25, 314, 245, 329 },
  })
}

\newcommand\kachoAnimBox[2]{%
  \hbox to 220bp{\vrule width 0pt height 280bp depth 0pt\directlua{anim.put(#1, #2)}\hss}%
}
\newcommand\kachoAnimFramePage[2]{%
  \noindent{\small\scshape Kach\=o-e Growth \hfill Frame #1/24}\par\vspace{4bp}%
  \noindent\kachoAnimBox{#2}{{ clip_box = true, frames_mode = true }}%
}
\providecommand\posterframe{last}

\ifdefined\kachoAnimFrames\def\framespages{1}\fi
\directlua{anim.check_frames_env()}

\begin{document}
\ifdefined\framespages
  \directlua{anim.emit_frames_pages(24)}
\else
  \noindent{\small\scshape Kach\=o-e Procedural Growth}\par\vspace{4bp}%
  \noindent
  \edef\kachoAnimBegin{\noexpand\begin{animateinline}[controls,loop,autoplay,poster=\posterframe]}\kachoAnimBegin{12}%
    \multiframe{24}{rT=0+0.04347826}{\kachoAnimBox{math.min(1,\rT)}{{}}}%
  \end{animateinline}%
\fi
\end{document}
